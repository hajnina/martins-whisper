        val db = amplitudeDb(AudioMath.rms(frame))
        val multiplierDb = if (thresholdMultiplier > 1.0) {
            20.0 * log10(thresholdMultiplier)
        } else {
            0.0
        }
        val thresholdDb = max(
            noiseFloorDb + 6.0,
            max(minSpeechRmsDb, noiseFloorDb + multiplierDb),
        )
        val speech = db > thresholdDb

        if (speech) {
            hangoverRemaining = hangoverFrames.coerceAtLeast(0)
        } else {
            noiseFloorDb += noiseFloorAlpha * (db - noiseFloorDb)
            if (hangoverRemaining > 0) {
                hangoverRemaining--
                return true
            }
        }
        return speech
    }

    override fun reset() {
        noiseFloorDb = -90.0
        hangoverRemaining = 0
    }

    private fun amplitudeDb(rms: Double): Double =
        if (rms <= 0.0) -120.0 else 20.0 * log10(rms)

    internal fun noiseFloorForTest(): Double = noiseFloorDb

    @Suppress("unused")
    private fun linearToDb(value: Double): Double = 20.0 * log10(value.coerceAtLeast(1e-12))

    @Suppress("unused")
    private fun dbToLinear(db: Double): Double = 10.0.pow(db / 20.0)
}

package com.martins.whisper.core.text

object TextNormalizer {
    private val hallucinations = setOf(
        "thank you.",
        "thank you",
        "thank you for watching.",
        "thank you for watching",
        "thanks for watching.",
        "thanks for watching!",
        "thanks for watching",
        "please subscribe.",
        "please subscribe",
        "[blank_audio]",
        "[ blank_audio ]",
        "[blank audio]",
        "[ blank audio ]",
        "[silence]",
        "[ silence ]",
        "(silence)",
        "*silence*",
        "[music]",
        "[ music ]",
        "[applause]",
        "[laughter]",
        "(upbeat music)",
        "...",
        "you",
    )

    private val whitespace = Regex("\\s+")
    private val spaceBeforePunctuation = Regex("\\s+([,.;:!?])")

    fun normalize(text: String): String {
        var value = text.replace(whitespace, " ").trim()
        value = spaceBeforePunctuation.replace(value, "$1")
        if (value.isBlank()) return ""
        if (hallucinations.contains(value.lowercase())) return ""
        return value
    }
}
